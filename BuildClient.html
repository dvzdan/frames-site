function hydrateBuildCopy() { const plans = SITE_CONTENT.sections.plans || {}; const updates = [ ["#build-print .build-step-text", plans.printText] ]; updates.forEach(function(update) { const node = document.querySelector(update[0]); if (node && update[1]) node.textContent = update[1]; }); } function populateBuildParts() { const parts = SITE_CONTENT.sections.parts || {}; const table = document.getElementById("partsTable"); const note = document.getElementById("partsNote"); if (!table) return; table.innerHTML = ""; (parts.items || []).forEach(function(item) { const itemName = item && (item.item || item.name); if (!itemName) return; const row = el("details", "parts-row"); const summary = el("summary", "parts-row-summary"); summary.appendChild(el("div", "parts-name", itemName)); row.appendChild(summary); const facts = el("div", "parts-facts"); appendBuildPartFact(facts, parts.productLabel || "Product", item.product, item.url); appendBuildPartFact(facts, parts.quantityLabel || "Quantity", item.quantity); appendBuildPartFact(facts, "Note", item.note); if (facts.children.length) row.appendChild(facts); table.appendChild(row); }); if (note) note.textContent = parts.note || ""; } function appendBuildPartFact(parent, label, value, url) { if (!value) return; const fact = el("div", "parts-fact"); fact.appendChild(el("span", "parts-fact-label", label)); if (url) { const link = el("a", "parts-link", value); link.href = url; link.target = "_blank"; link.rel = "noopener"; fact.appendChild(link); } else { const content = el("span", "parts-fact-value"); content.appendChild(createTermTooltip(value)); fact.appendChild(content); } parent.appendChild(fact); } function startBuildPage() { hydrateBuildCopy(); populateBuildParts(); bindTermHelp(); bindSupportGuideModal(); } if (document.readyState === "loading") { document.addEventListener("DOMContentLoaded", startBuildPage); } else { startBuildPage(); }
