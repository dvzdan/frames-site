function hydrateBuildCopy() { const plans = SITE_CONTENT.sections.plans || {}; const updates = [ ["#build-print .build-step-heading", plans.printHeading], ["#build-print .build-step-text", plans.printText], ["#build-hardware .build-step-heading", plans.hardware && plans.hardware.heading], ["#build-hardware .build-step-text", plans.hardware && plans.hardware.text], ["#build-continue .build-step-heading", plans.continue && plans.continue.heading], ["#build-continue .build-step-text", plans.continue && plans.continue.text] ]; updates.forEach(function(update) { const node = document.querySelector(update[0]); if (node && update[1]) node.textContent = update[1]; }); const hardwareLink = document.querySelector("#build-hardware .plans-hardware-link"); if (hardwareLink && plans.hardware && plans.hardware.linkLabel) { hardwareLink.textContent = plans.hardware.linkLabel; } const continueLink = document.querySelector("#build-continue .plans-continue-link"); if (continueLink && plans.continue && plans.continue.linkLabel) { continueLink.textContent = plans.continue.linkLabel; } } function enhanceBuildDownloads() { document.querySelectorAll(".download-type-list dd").forEach(function(node) { const value = node.textContent.trim(); node.replaceChildren(createTermTooltip(value)); }); const list = document.getElementById("downloadsList"); const printCard = document.getElementById("build-print"); if (!list || !printCard) return; const rows = Array.prototype.slice.call(list.querySelectorAll(".download-row")); const primary = rows.find(function(row) { const name = row.querySelector(".download-name"); return name && /\.3mf\b/i.test(name.textContent || ""); }); if (!primary) return; const name = primary.querySelector(".download-name"); primary.className = "button build-primary-action build-primary-download"; primary.setAttribute("aria-label", "Download " + (name ? name.textContent.trim() : "3MF print file")); primary.textContent = "Download 3MF"; const heading = printCard.querySelector(".plans-downloads-heading"); if (heading) heading.insertAdjacentElement("afterend", primary); } function populateBuildParts() { const parts = SITE_CONTENT.sections.parts || {}; const table = document.getElementById("partsTable"); const note = document.getElementById("partsNote"); if (!table) return; table.innerHTML = ""; (parts.items || []).forEach(function(item) { const itemName = item && (item.item || item.name); if (!itemName) return; const row = el("details", "parts-row"); const summary = el("summary", "parts-row-summary"); summary.appendChild(el("div", "parts-name", itemName)); row.appendChild(summary); const facts = el("div", "parts-facts"); appendBuildPartFact(facts, parts.productLabel || "Product", item.product, item.url); appendBuildPartFact(facts, parts.quantityLabel || "Quantity", item.quantity); appendBuildPartFact(facts, "Note", item.note); if (facts.children.length) row.appendChild(facts); table.appendChild(row); }); if (note) note.textContent = parts.note || ""; } function appendBuildPartFact(parent, label, value, url) { if (!value) return; const fact = el("div", "parts-fact"); fact.appendChild(el("span", "parts-fact-label", label)); if (url) { const link = el("a", "parts-link", value); link.href = url; link.target = "_blank"; link.rel = "noopener"; fact.appendChild(link); } else { const content = el("span", "parts-fact-value"); content.appendChild(createTermTooltip(value)); fact.appendChild(content); } parent.appendChild(fact); } function startBuildPage() { hydrateBuildCopy(); enhanceBuildDownloads(); populateBuildParts(); bindTermHelp(); bindSupportGuideModal(); } if (document.readyState === "loading") { document.addEventListener("DOMContentLoaded", startBuildPage); } else { startBuildPage(); }
